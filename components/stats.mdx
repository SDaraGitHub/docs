## With PrimeNG components

```typescript
import { Component } from '@angular/core';
import { CardModule } from 'primeng/card';

@Component({
  selector: 'app-stats',
  template: `
    <div class="surface-ground px-4 py-8 md:px-6 lg:px-8">
      <div class="grid">
        <div class="col-12 md:col-6 lg:col-3" *ngFor="let stat of stats">
          <p-card styleClass="text-center">
            <i [class]="'pi ' + stat.icon + ' text-5xl mb-3'" [style.color]="stat.color"></i>
            <div class="text-900 font-bold text-4xl mb-2">{{ stat.value }}</div>
            <div class="text-700 text-xl">{{ stat.label }}</div>
          </p-card>
        </div>
      </div>
    </div>
  `,
  standalone: true,
  imports: [CardModule]
})
export class StatsComponent {
  stats = [
    { icon: 'pi-users', value: '10K+', label: 'Active Users', color: '#3B82F6' },
    { icon: 'pi-globe', value: '50+', label: 'Countries', color: '#10B981' },
    { icon: 'pi-chart-line', value: '99.9%', label: 'Uptime', color: '#8B5CF6' },
    { icon: 'pi-star', value: '4.9/5', label: 'Rating', color: '#F59E0B' }
  ];
}
```
